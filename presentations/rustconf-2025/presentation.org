#+BEGIN_SRC rust
pub struct ContextManager {
    history: Vec<Message>,
    token_counter: TokenCounter,
    session_store: SessionStore,
    max_context: usize,
}

impl ContextManager {
    pub fn add_message(&mut self, msg: Message) -> Result<()> {
        self.prune_if_needed()?;
        self.history.push(msg);
        self.persist()?;
        Ok(())
    }
}
#+END_SRC

#+BEGIN_NOTES
- Token counting varies by model
- Implemented sliding window with importance scoring
- Sessions auto-save every N messages
- Can resume conversations after crashes

#+BEGIN_SRC rust
pub struct SecuritySandbox {
    workspace: PathBuf,
    allowed_operations: HashSet<Operation>,
}

impl SecuritySandbox {
    pub fn validate_path(&self, path: &Path) -> Result<()> {
        let canonical = path.canonicalize()?;
        if !canonical.starts_with(&self.workspace) {
            return Err(SecurityError::PathTraversal);
        }
        Ok(())
    }
}
#+END_SRC

#+BEGIN_SRC rust
#[derive(Debug, thiserror::Error)]
pub enum GeminiError {
    #[error("API error: {0}")]
    Api(#[from] ApiError),
    
    #[error("Tool execution failed: {0}")]
    Tool(#[from] ToolError),
    
    #[error("Security violation: {0}")]
    Security(#[from] SecurityError),
}

// Automatic conversion from subsystem errors
// No unwrap() needed - ever!
#+END_SRC

use gemini_repl::{Repl, Config};

#[tokio::main]
async fn main() -> Result<()> {
    let config = Config::from_env()?;
    let mut repl = Repl::new(config).await?;
    
    // Self-modification enabled!
    repl.enable_plugin_generation();
    repl.set_workspace("./workspace")?;
    
    repl.run().await?;
    Ok(())
}
